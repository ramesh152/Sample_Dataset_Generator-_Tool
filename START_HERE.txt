🎯 WHAT THIS IS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
A tool to generate synthetic datasets for testing Vision-Language Models:

For each sample, creates:
  ✓ input.jpg    → Original image with source text
  ✓ expected.jpg → Perfect translated/processed version
  ✓ A.jpg        → Slightly flawed (minor issues)
  ✓ B.jpg        → Best quality (near perfect)
  ✓ C.jpg        → Bad quality (clearly wrong)
  ✓ meta.json    → Labels and metadata

Generates 10 to 10,000 images with controlled variations at scale.

🎬 STEP-BY-STEP
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
STEP 1: Install Dependencies
  $ pip install -r generator_requirements.txt
  (Takes 1-2 minutes)

STEP 2: Start the Streamlit App
  $ streamlit run auto_dataset_streamlit.py
  (Browser opens automatically at http://localhost:8501)

STEP 3: Go to "🎯 Generate Dataset" Tab
  (It's the first tab - selected by default)

STEP 4: Configure Generation
  • Number of Samples: 100 (good starting point)
  • Image Width: 800 (or 512 for faster)
  • Image Height: 600 (or 512 for faster)
  • Base Quality: High
  • Variation Strength: 0.7 (balanced)

STEP 5: Click "🎬 Generate Dataset"
  (Wait for generation - status shown in progress bar)

STEP 6: Review Dataset
  • Go to "📊 Dataset Info" tab
  • Browse samples using slider
  • Check metadata

STEP 7: Download or Manage
  • Go to "📥 Manage Datasets" tab
  • See all generated datasets
  • Download or delete as needed
